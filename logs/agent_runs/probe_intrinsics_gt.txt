def main():
    scene_id = "probe_scene"

    res = list_objects(scene_id=scene_id)
    if not res.ok:
        submit("unknown", evidence=["list_objects failed: %s" % res.error])
        return

    objs = res.value or []
    if not objs:
        submit("unknown", evidence=["scene has no objects"])
        return

    best = None
    best_d = None
    for o in objs:
        c = o["centroid_m"]

    label = best["label"]
    oid = best["object_id"]
    submit(
        label,
        target_ids=[oid],
        evidence=[
            "list_objects(scene_id=%r) returned %d objects with centroid_m" % (scene_id, len(objs)),
